# Run by Dovecot's IMAPSieve when a user copies a message back out of the
# Spam (or Junk) folder. Pipes the message to the sa-learn wrapper so that
# SpamAssassin's bayes database learns it as ham.
#
# Deleting spam is a move from Spam to Trash, which must not be learned as
# ham. The old dovecot-antispam plugin expressed that as
# antispam_trash_pattern_ignorecase = trash;Deleted *
#
# Installed by setup/spamassassin.sh as /usr/lib/dovecot/sieve/report-ham.sieve.

require ["vnd.dovecot.pipe", "copy", "imapsieve", "environment", "variables"];

if environment :matches "imap.mailbox" "*" {
  set "mailbox" "${1}";
}

if anyof (string :is "${mailbox}" "Trash",
          string :matches "${mailbox}" "Deleted*") {
  stop;
}

if environment :matches "imap.user" "*" {
  set "username" "${1}";
}

pipe :copy "sa-learn-ham.sh" [ "${username}" ];
