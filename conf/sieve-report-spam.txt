# Run by Dovecot's IMAPSieve when a user moves or copies a message into the
# Spam (or Junk) folder. Pipes the message to the sa-learn wrapper so that
# SpamAssassin's bayes database learns from it.
#
# Installed by setup/spamassassin.sh as /usr/lib/dovecot/sieve/report-spam.sieve.

require ["vnd.dovecot.pipe", "copy", "imapsieve", "environment", "variables"];

if environment :matches "imap.user" "*" {
  set "username" "${1}";
}

pipe :copy "sa-learn-spam.sh" [ "${username}" ];
